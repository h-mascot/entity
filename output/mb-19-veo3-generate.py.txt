#!/usr/bin/env python3
"""
Minimal Veo 3 generator for Henry's content pipeline.

Usage:
  python3 veo3_generate.py --prompt "..." [--image path] [--seconds 8] [--out name]

Saves output to ~/clawd/output/videos/<name>.mp4 and metadata JSON sidecar.
"""

from __future__ import annotations

import argparse
import base64
import json
import mimetypes
import os
import pathlib
import sys
import time
from datetime import datetime


def _env_api_key() -> str | None:
    return os.getenv("GOOGLE_API_KEY") or os.getenv("GEMINI_API_KEY")


def _load_image_inline(path: pathlib.Path) -> dict:
    mime = mimetypes.guess_type(str(path))[0] or "image/png"
    b64 = base64.b64encode(path.read_bytes()).decode("utf-8")
    return {"mime_type": mime, "data": b64}


def main() -> int:
    p = argparse.ArgumentParser(description="Generate videos with Veo 3 via Google GenAI")
    p.add_argument("--prompt", required=True, help="Video prompt")
    p.add_argument("--image", help="Optional image path for image-conditioned generation")
    p.add_argument("--seconds", type=int, default=8, help="Requested duration in seconds")
    p.add_argument("--model", default="veo-3.0-generate-001", help="Veo model id")
    p.add_argument("--poll-seconds", type=int, default=10, help="Polling interval")
    p.add_argument("--timeout-seconds", type=int, default=900, help="Max wait time")
    p.add_argument("--out", help="Output basename (without extension)")
    args = p.parse_args()

    key = _env_api_key()
    if not key:
        print("ERROR: GOOGLE_API_KEY or GEMINI_API_KEY is required", file=sys.stderr)
        return 2

    try:
        from google import genai
    except Exception:
        print("ERROR: google-genai is not installed. Run: pip install google-genai", file=sys.stderr)
        return 2

    output_dir = pathlib.Path("~/clawd/output/videos").expanduser()
    output_dir.mkdir(parents=True, exist_ok=True)

    ts = datetime.now().strftime("%Y%m%d-%H%M%S")
    basename = args.out or f"veo3-{ts}"
    video_path = output_dir / f"{basename}.mp4"
    meta_path = output_dir / f"{basename}.json"

    client = genai.Client(api_key=key)

    generation_config = {
        "duration_seconds": args.seconds,
    }

    # NOTE: API signatures can evolve; this call is intentionally minimal + tolerant.
    payload = {
        "model": args.model,
        "prompt": args.prompt,
        "config": generation_config,
    }

    if args.image:
        image_path = pathlib.Path(args.image).expanduser().resolve()
        if not image_path.exists():
            print(f"ERROR: image file not found: {image_path}", file=sys.stderr)
            return 2
        payload["image"] = _load_image_inline(image_path)

    try:
        op = client.models.generate_videos(**payload)
    except Exception as e:
        print("ERROR: generate_videos request failed.", file=sys.stderr)
        print(str(e), file=sys.stderr)
        return 1

    start = time.time()
    while not getattr(op, "done", False):
        if time.time() - start > args.timeout_seconds:
            print("ERROR: Timed out waiting for video generation", file=sys.stderr)
            return 1
        time.sleep(args.poll_seconds)
        try:
            op = client.operations.get(op)
        except Exception as e:
            print("ERROR: polling operation failed", file=sys.stderr)
            print(str(e), file=sys.stderr)
            return 1

    # Pull bytes from first generated video
    try:
        generated = op.response.generated_videos[0]
        video_blob = client.files.download(file=generated.video)
        video_path.write_bytes(video_blob)
    except Exception as e:
        print("ERROR: failed to download/write generated video", file=sys.stderr)
        print(str(e), file=sys.stderr)
        return 1

    meta = {
        "created_at": datetime.now().isoformat(),
        "model": args.model,
        "prompt": args.prompt,
        "image": str(args.image) if args.image else None,
        "seconds": args.seconds,
        "operation_done": True,
        "video_path": str(video_path),
    }
    meta_path.write_text(json.dumps(meta, indent=2), encoding="utf-8")

    print(json.dumps({"ok": True, "video": str(video_path), "meta": str(meta_path)}, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
